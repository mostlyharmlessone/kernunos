        COMPLEX (wp), INTENT (IN) :: z
        INTEGER, OPTIONAL :: flags(3)
        COMPLEX (wp) :: abi(4)

        INTEGER ierra, ierrb, ierrc
        LOGICAL :: argument_switch = .FALSE., &
                   modify_switch = .FALSE.
        COMPLEX (wp) alpha, beta, gam  , bi1(2), bi2(2)
        REAL (wp) :: half = 0.5E0_wp, three = 3.0E0_wp

! From ACM-TOMS 838:
        CALL airy_ai(z, abi(1), abi(2), ierra, &
                     argument_switch, modify_switch)
! Use connection formulas for complex values to get Bi(z), Bi'(z). 
! Bi(z) = exp(-pi * i /6) * Ai(z * exp (-2 * pi * i /3)) +
!         exp( pi * i /6) * Ai(z * exp ( 2 * pi * i /3))      

! Note: Specify wp in CMPLX intrinsic  or else it will default 
! to single precision for the double precision code..
        beta = -CMPLX(half, SQRT(three)*half, wp)

        CALL airy_ai(z*beta, bi1(1), bi1(2), ierrb, &
                     argument_switch, modify_switch)
        CALL airy_ai(z*CONJG(beta), bi2(1), bi2(2), ierrc, &
                     argument_switch, modify_switch)
        alpha = CMPLX(sqrt(three)*half, -half, wp)
        gam   = -CONJG(alpha)
        abi(3) = alpha*bi1(1) + CONJG(alpha)*bi2(1)
        abi(4) = gam  *bi1(2) + CONJG(gam  )*bi2(2)
        IF (PRESENT(flags)) THEN
          flags(1) = ierra
          flags(2) = ierrb
          flags(3) = ierrc
          RETURN
        END IF
        IF (ierra==0 .AND. ierrb==0 .AND. ierrc==0) RETURN
