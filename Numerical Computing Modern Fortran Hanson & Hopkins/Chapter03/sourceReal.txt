        REAL (wp), INTENT (IN) :: x
        INTEGER, OPTIONAL :: flags(2)
        REAL (wp) :: abi(4)

        INTEGER :: ierra, ierrb
        LOGICAL :: modify_switch = .FALSE.

! From ACM-TOMS 838:
        CALL airy_ai(x, abi(1), abi(2), ierra, modify_switch)
        CALL airy_bi(x, abi(3), abi(4), ierrb, modify_switch)
        IF (PRESENT(flags)) THEN
          flags(1) = ierra
          flags(2) = ierrb
          RETURN
        END IF
        IF (ierra==0 .AND. ierrb==0) RETURN
